cmake_minimum_required(VERSION 3.22)
project(MV2Recreation VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
include(CTest)
add_executable(dsp_tests tests/DynamicsTests.cpp)
target_include_directories(dsp_tests PRIVATE Source)
add_test(NAME dynamics COMMAND dsp_tests)
option(MV2_BUILD_PLUGIN "Build JUCE plugin and standalone" ON)
if(MV2_BUILD_PLUGIN)
  include(FetchContent)
  option(MV2_USE_INSTALLED_JUCE "Use a preinstalled JUCE SDK (including native juceaide for cross builds)" OFF)
  if(MV2_USE_INSTALLED_JUCE)
    find_package(JUCE 8.0.6 EXACT CONFIG REQUIRED)
  elseif(DEFINED JUCE_SOURCE_DIR)
    add_subdirectory("${JUCE_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/JUCE")
  else()
    FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 51a8a6d7aeae7326956d747737ccf1575e61e209)
    FetchContent_MakeAvailable(JUCE)
  endif()
  juce_add_binary_data(MV2Assets SOURCES assets/mv2-reference.png)
  set(mv2_manifest TRUE)
  if(CMAKE_CROSSCOMPILING)
    set(mv2_manifest FALSE)
  endif()
  juce_add_plugin(MV2Recreation COMPANY_NAME "Aulamix" BUNDLE_ID "com.aulamix.mv2recreation"
    PLUGIN_MANUFACTURER_CODE Aumx PLUGIN_CODE Mv2r
    FORMATS VST3 Standalone PRODUCT_NAME "MV2 Recreation"
    IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE
    COPY_PLUGIN_AFTER_BUILD FALSE VST3_AUTO_MANIFEST ${mv2_manifest})
  if(MSVC)
    target_compile_options(MV2Recreation PRIVATE /utf-8)
  endif()
  juce_generate_juce_header(MV2Recreation)
  target_sources(MV2Recreation PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
  target_compile_definitions(MV2Recreation PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0 JUCE_DISPLAY_SPLASH_SCREEN=0)
  target_link_libraries(MV2Recreation PRIVATE MV2Assets juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags)
endif()
if(MV2_BUILD_PLUGIN AND BUILD_TESTING)
  add_executable(processor_tests tests/ProcessorTests.cpp)
  target_include_directories(processor_tests PRIVATE Source "${CMAKE_BINARY_DIR}/MV2Recreation_artefacts/JuceLibraryCode" "$<TARGET_PROPERTY:juce::juce_core,INTERFACE_INCLUDE_DIRECTORIES>")
  target_compile_definitions(processor_tests PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
  target_link_libraries(processor_tests PRIVATE MV2Recreation MV2Assets)
  add_test(NAME processor COMMAND processor_tests)
endif()
